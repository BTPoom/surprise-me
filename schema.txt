generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

model Account {
  id                String  @id @default(cuid())
  userId            String
  type              String
  provider          String
  providerAccountId String
  refresh_token     String? @db.Text
  access_token      String? @db.Text
  expires_at        Int?
  token_type        String?
  scope             String?
  id_token          String? @db.Text
  session_state     String?
  user              User    @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@unique([provider, providerAccountId])
}

model Session {
  id           String   @id @default(cuid())
  sessionToken String   @unique
  userId       String
  expires      DateTime
  user         User     @relation(fields: [userId], references: [id], onDelete: Cascade)
}

model User {
  id            String    @id @default(cuid())
  name          String?
  email         String    @unique
  emailVerified DateTime?
  image         String?
  password      String?   // For credentials provider
  accounts      Account[]
  sessions      Session[]
  pages         Page[]
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt
}

model VerificationToken {
  identifier String
  token      String   @unique
  expires    DateTime

  @@unique([identifier, token])
}

model Page {
  id          String     @id @default(cuid())
  slug        String     @unique
  title       String
  message     String     @db.Text
  senderName  String
  occasion    String     // birthday, anniversary, graduation, custom
  theme       String     @default("rose")
  youtubeUrl  String?
  youtubeId     String?
  youtubeStartAt Int?       @default(0)
  youtubeEndAt   Int?
  password    String?    // bcrypt hashed
  scheduledAt DateTime?
  expiresAt   DateTime?
  status      String     @default("draft") // draft | published | expired
  photos      Photo[]
  reactions   Reaction[]
  analytics   Analytics[]
  owner       User       @relation(fields: [ownerId], references: [id], onDelete: Cascade)
  ownerId     String
  createdAt   DateTime   @default(now())
  updatedAt   DateTime   @updatedAt

  @@index([slug])
  @@index([ownerId])
  @@index([status])
}

model Photo {
  id        String   @id @default(cuid())
  url       String
  caption   String?
  order     Int      @default(0)
  page      Page     @relation(fields: [pageId], references: [id], onDelete: Cascade)
  pageId    String
  createdAt DateTime @default(now())
}

model Reaction {
  id        String   @id @default(cuid())
  emoji     String
  message   String?  @db.Text
  createdAt DateTime @default(now())
  page      Page     @relation(fields: [pageId], references: [id], onDelete: Cascade)
  pageId    String

  // Privacy-first: no IP, no fingerprint, no userId
  @@index([pageId])
}

model Analytics {
  id       String   @id @default(cuid())
  page     Page     @relation(fields: [pageId], references: [id], onDelete: Cascade)
  pageId   String
  openedAt DateTime @default(now())
  // Privacy-first: no IP, no userAgent, no fingerprint

  @@index([pageId])
  @@index([openedAt])
}
